automacro HP_Baixo {
    CurrentHP < 90%
    exclusive 1
    timeout 15
    call move_to_sacer 
}

automacro HP_subiu {
    CurrentHP > 90%
    timeout 10
    exclusive 1
    call RESTAURAR_CONFIG
}

macro move_to_sacer  {
    do conf follow 1
    do conf followBot 1
    do conf followTarget Romana
}

macro RESTAURAR_CONFIG {
    do conf follow 0
    do conf followBot 0
}
